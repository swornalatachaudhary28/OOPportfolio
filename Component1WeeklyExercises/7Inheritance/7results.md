package Week7;

public  class Playerstatistics {
	protected String playerName;
    protected String teamName;

    public Playerstatistics(String name, String team) {
        this.playerName = name;
        this.teamName = team;
    }

    public void displayBasicInfo() {
        System.out.print(playerName + " (" + teamName + "): ");
    }

}

package Week7;

public class Cricket extends Playerstatistics{
	private int wickets;
	
	public Cricket(String name, String team, int wickets) {
        super(name, team); 
        this.wickets = wickets;
    }

    public void displayStats() {
        displayBasicInfo(); 
        System.out.println( wickets + " wickets taken.");
    }
}

package Week7;

public class Football extends Playerstatistics{
	private int goals;
	public Football(String name, String team, int goals) {
        super(name, team); 
        this.goals = goals;
    }

    public void displayStats() {
        displayBasicInfo(); 
        System.out.println(goals + " goals scored.");
    }
}
