package Week8;

public interface Encryptable {
	public void encrypt();
    public String decrypt();
}

package Week8;

public class Secret implements Encryptable {
	private String message;
    private boolean encrypted;

    public Secret(String msg) {
        message = msg;
        encrypted = false;
    }

    public void encrypt() {
        if (!encrypted) {
            String masked = "";
            for (int i = 0; i < message.length(); i++) {
                masked = masked + (char) (message.charAt(i) + 3);
            }
            message = masked;
            encrypted = true;
        }
    }

    public String decrypt() {
        if (encrypted) {
            String unmasked = "";
            for (int i = 0; i < message.length(); i++) {
                unmasked = unmasked + (char) (message.charAt(i) - 3);
            }
            message = unmasked;
            encrypted = false;
        }
        return message;
    }

    public String toString() {
        return message;
    }
}

package Week8;

public class Password implements Encryptable {
	private String key;
    private boolean encrypted;

    public Password(String pass) {
        key = pass;
        encrypted = false;
    }

    public void encrypt() {
        if (!encrypted) {
            // Reverses the string
            key = new StringBuilder(key).reverse().toString();
            encrypted = true;
        }
    }

    public String decrypt() {
        if (encrypted) {
            key = new StringBuilder(key).reverse().toString();
            encrypted = false;
        }
        return key;
    }

    public String toString() {
        return key;
    }
}
